import java.io.*;
import java.util.*;

public class Solution {
	
	static int[][] numbers;
	static int[][] sum;
	
	public static void main(String[] args) throws IOException {
		
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringBuilder sb = new StringBuilder();
		
		int T = Integer.parseInt(br.readLine());
		for(int test_case = 1; test_case <= T; test_case++) {
			int N = Integer.parseInt(br.readLine());
			numbers = new int[N][];
			sum = new int[N][];
			
			for(int i = 0; i < N; i++) {
				numbers[i] = new int[i + 1];
				sum[i] = new int[i + 1];
				StringTokenizer st = new StringTokenizer(br.readLine());
				for(int j = 0; j < numbers[i].length; j++) {
					numbers[i][j] = Integer.parseInt(st.nextToken());
				}
			}

			int maxSum = numbers[0][0];
			sum[0][0] = numbers[0][0];
			for(int i = 1; i < N; i++) {
				sum[i][0] = sum[i - 1][0] + numbers[i][0];
				sum[i][i] = sum[i - 1][i - 1] + numbers[i][i];
				maxSum = Math.max(maxSum, sum[i][0]);
				maxSum = Math.max(maxSum, sum[i][i]);
			}
			
			for(int i = 2; i < N; i++) {
				for(int j = 1; j < i; j++) {
					sum[i][j] = numbers[i][j] + Math.max(sum[i - 1][j - 1], sum[i - 1][j]);
					maxSum = Math.max(maxSum, sum[i][j]);
				}
			}
			
			sb.append("#").append(test_case).append(" ");
			sb.append(maxSum);
			sb.append("\n");
		}
		
		System.out.println(sb);
	}
}
